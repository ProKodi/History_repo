findall(GC, grandchild(GC, _), AllGrandchildren).
findall(GC, grandson(GC, _), AllGrandchildren).

findall(GC, grandson(GC, _), AllGrandchildren),
foreach(member(GC, AllGrandchildren), 
        format('Внук: ~w~n', [GC])).
