# Описание алгоритма по шагам
Алгоритм выбирает один элемент массива (опорный) и разделяет остальные элементы на две группы: 
- меньше опорного
- больше него
Затем те же действия рекурсивно выполняются для полученных подмассивов. <br>

Общая идея алгоритма состоит в следующем:
1) Выбрать из массива элемент, называемый опорным. <br>
    Это может быть любой из элементов массива. <br>
    От выбора опорного элемента не зависит корректность алгоритма,
        но в отдельных случаях может сильно зависеть его эффективность

2) Сравнить все остальные элементы с опорным и переставить их в массиве так, 
    чтобы разбить массив на три непрерывных отрезка, 
    следующих друг за другом: <br>
    1) «элементы меньшие опорного»
    2) «равные» 
    3) «большие»

4) Для отрезков «меньших» и «больших» значений выполнить рекурсивно 
    ту же последовательность операций, если длина отрезка больше единицы

## Шаги алогоритма:
1) Выбрать опорный элемент (pivot).<br>
    Чаще всего выбирают:
    - первый элемент
    - последний элемент
    - средний элемент
    - случайный элемент

2) Разделить массив на две части:
   1) элементы **меньше pivot**
   2) элементы **больше pivot**

3) Поместить pivot **между этими частями**.
4) Когда подмассив содержит **0 или 1 элемент**, он считается отсортированным.
5) Рекурсивно применить алгоритм к полученым алгоритмам


## Оценка по сложности и по памяти
### Сложность: 
- Лучший/Средний случай: **O(n * log(n))**
- Худший случай: **O(n^2)**
  
### Затраты по памяти
- **O(log(n))** - при условии что компилятор/интерпретатор не заменит рекурсию на цикл
- **O(1)** - при условии что компилятор/интерпретатор заменит рекурсию на цикл

## Шаг 1 — выбираем pivot
pivot = **7**

pivot = **2** <br>
Левая (меньше пивота): [] (``коллекция отсортирована``) <br>
Правая (больше пивота): [4, 3] <br>
Коллекция в целом: [2, 4, 3, 7, 10, 9] <br>

pivot = **4** <br>

Коллекция в целом: [2, 3, 4, 7, 10, 9] <br>


## Шаг 4 - Сортируем правую часть
pivot = **10** <br>
Левая (меньше пивота): [9] (``коллекция отсортирована``) <br>
Правая (больше пивота): [] (``коллекция отсортирована``) <br>
Коллекция в целом: [2, 4, 3, 7, 9, 10] <br>

## Python
```py
def bad_quick_sort(arr):
    """ Алгоритм с излишним использование памяти """
    if len(arr) <= 1:
        return arr

    pivot = arr[0]

    left = []
    right = []
    lst_pivon = []
    
    
    for i in arr:
        if(i < pivot): 
            left.append(i)

        elif(i > pivot):
            right.append(i)

        else: lst_pivon.append(i)

    return bad_quick_sort(left) + lst_pivon + bad_quick_sort(right)


data = [7, 2, 9, 4, 3, 10]
print(bad_quick_sort(data))


def normal_quick_sort(arr, min_index = None, max_index = None):
    # Если не указаны индексы
    if((min_index == None) or (max_index == None)):
        min_index = 0
        max_index = len(arr) - 1

    # Если сортируемая коллекция представляет из себя коллекцию с 1 элементом или без элементов
    if(max_index - min_index < 1): return

    # Индекс пивота
    pivot: int = min_index - 1
    
    # Сравниваем коллекцию с последним элементом (последний)
    # Левые элементы меньше последнего, правые больше/равны последнему
    for i in range(min_index, max_index):
        # Если текущий элемент меньше
        if(arr[i] < arr[max_index]):
            # Двигаем пивот в право (левая коллекция уверичивается на 1)
            pivot += 1
            # текущий элемент становится левым, пивот - правым
            arr[pivot], arr[i] = arr[i], arr[pivot]

    # После цикла слева (включая пивот) меньше последнего, справа - больше/равные 

    # двигаем пивот в право
    pivot += 1

    # последний устанавливаем на пивот, пивот на последний
    arr[pivot], arr[max_index] = arr[max_index], arr[pivot]

    # Сортируем левую коллекцию
    normal_quick_sort(arr, min_index, pivot - 1);
    # Сортируем правую коллекцию
    normal_quick_sort(arr, pivot + 1, max_index);


data = [7, 2, 9, 4, 3, 10]
normal_quick_sort(data)
print(data)
```

```cs
class Program{
    public static void BabQuickSort(int[] arr, int left, int right){
        int i = left;
        int j = right;
        int pivot = arr[(left + right) / 2];

        while (i <= j){
            while (arr[i] < pivot) { i++; }
            while (arr[j] > pivot) { j--; }
                
            if (i <= j){
                int temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;

                i++;
                j--;
            }
        }
        if (left < j){ BabQuickSort(arr, left, j); }
        if (i < right){ BabQuickSort(arr, i, right); }
    }


    public static void NormalQuickSort(int[] arr){ NormalQuickSort(arr, 0, arr.Count() - 1); }

    protected static void NormalQuickSort(int[] arr, int min_index, int max_index){
        // Если сортируемая коллекция представляет из себя коллекцию с 1 элементом или без элементов
        if(max_index - min_index < 1){ return; } 

        // Индекс пивота
        int pivot = min_index - 1;

        // Сравниваем коллекцию с последним элементом (последний)
        // Левые элементы меньше последнего, правые больше/равны последнему
        for(int i = min_index; i < max_index; i += 1){
            // Если текущий элемент меньше
            if(arr[i] < arr[max_index]){
                // Двигаем пивот в право (левая коллекция уверичивается на 1)
                pivot += 1;
                // текущий элемент становится левым, пивот - правым
                (arr[pivot], arr[i]) = (arr[i], arr[pivot]);
            }
        }
        // После цикла слева (включая пивот) меньше последнего, справа - больше/равные 
        // двигаем пивот в право
        pivot += 1;

        // последний устанавливаем на пивот, пивот на последний
        (arr[pivot], arr[max_index]) = (arr[max_index], arr[pivot]);

        Parallel.Invoke(
            // Сортируем левую коллекцию
            () => { NormalQuickSort(arr, min_index, pivot - 1); },
            // Сортируем правую коллекцию
            () => { NormalQuickSort(arr, pivot + 1, max_index); }
        );
    }


    static void Main(){
        int[] arr = [7, 2, 9, 4, 3];
        BabQuickSort(arr, 0, arr.Length - 1);
        Console.WriteLine(string.Join(", ", arr));

        arr = [7, 2, 9, 4, 3];
        NormalQuickSort(arr);
        Console.WriteLine(string.Join(", ", arr));
    }
}
```
